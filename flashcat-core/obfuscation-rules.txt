# Obfuscation rules consumed by apps that depend on @flashcatcloud/core.
# Keep the public SDK entry points readable so integrators' stack traces and
# API calls aren't mangled. The SDK's own internals may still be obfuscated.

  Flashcat
  ConfigurationBuilder
  TrackingConsent
  FlashcatSite
  UploadFrequency
  BatchSize
  BatchProcessingLevel
-keep-property-name
  initialize
  initializeForDeferredUpload
  getInstance
  isInitialized
  setTrackingConsent
  setUserInfo
  clearUserInfo
  flush
  flushAndWait
  stopInstance
  setUploadFrequency
  setBatchSize
  setBatchProcessingLevel
